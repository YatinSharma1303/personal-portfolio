  2099→    function wrapCanvasText(ctx, text, maxWidth, maxLines) {
  2100→      const words = String(text || '').split(/\s+/);
  2101→      const lines = [];
  2102→      let line = '';
  2103→      for (let i = 0; i < words.length; i++) {
  2104→        const test = line ? line + ' ' + words[i] : words[i];
  2105→        if (ctx.measureText(test).width > maxWidth && line) {
  2106→          lines.push(line); line = words[i];
  2107→          if (lines.length === maxLines - 1) break;
  2108→        } else { line = test; }
  2109→      }
  2110→      if (lines.length < maxLines) lines.push(line);
  2111→      // Ellipsis if truncated.
  2112→      const used = lines.join(' ').split(/\s+/).length;
  2113→      if (used < words.length) lines[lines.length - 1] = (lines[lines.length - 1] || '').replace(/\s*\S*$/, '') + '…';
