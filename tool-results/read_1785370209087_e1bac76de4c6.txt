  2036→    function wrapCanvasText(ctx, text, maxWidth, maxLines) {
  2037→      const words = String(text || '').split(/\s+/);
  2038→      const lines = [];
  2039→      let line = '';
  2040→      for (let i = 0; i < words.length; i++) {
  2041→        const test = line ? line + ' ' + words[i] : words[i];
  2042→        if (ctx.measureText(test).width > maxWidth && line) {
  2043→          lines.push(line); line = words[i];
  2044→          if (lines.length === maxLines - 1) break;
  2045→        } else { line = test; }
  2046→      }
  2047→      if (lines.length < maxLines) lines.push(line);
  2048→      // Ellipsis if truncated.
  2049→      const used = lines.join(' ').split(/\s+/).length;
  2050→      if (used < words.length) lines[lines.length - 1] = (lines[lines.length - 1] || '').replace(/\s*\S*$/, '') + '…';
  2051→      return lines.filter(Boolean);
  2052→    }
