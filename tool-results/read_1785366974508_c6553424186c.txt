  1848→    function wrapCanvasText(ctx, text, maxWidth, maxLines) {
  1849→      const words = String(text || '').split(/\s+/);
  1850→      const lines = [];
  1851→      let line = '';
  1852→      for (let i = 0; i < words.length; i++) {
  1853→        const test = line ? line + ' ' + words[i] : words[i];
  1854→        if (ctx.measureText(test).width > maxWidth && line) {
  1855→          lines.push(line); line = words[i];
  1856→          if (lines.length === maxLines - 1) break;
  1857→        } else { line = test; }
  1858→      }
  1859→      if (lines.length < maxLines) lines.push(line);
  1860→      // Ellipsis if truncated.
  1861→      const used = lines.join(' ').split(/\s+/).length;
  1862→      if (used < words.length) lines[lines.length - 1] = (lines[lines.length - 1] || '').replace(/\s*\S*$/, '') + '…';
  1863→      return lines.filter(Boolean);
  1864→    }
